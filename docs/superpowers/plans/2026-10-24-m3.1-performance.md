  ```cpp
  // perf_stats.h (rawreel_core)
  namespace rawreel {
  struct ShowSample { std::int64_t index = 0; Clock::time_point at{}; };
  struct ShowTiming {
      double actualFps = 0;          // (n − 1) / (last − first)
      double p99LatenessMs = 0;      // nearest-rank p99 of the per-show lateness
      std::uint64_t reanchors = 0;   // shows more than one period behind the schedule (stalls)
  };
  // The schedule starts at the first show and advances by 1/fps per show; a show more than one
  // period late re-anchors it (as PlayerLogic does after a stall) and is not counted as lateness;
  // early shows count 0.
  [[nodiscard]] ShowTiming showTiming(std::span<const ShowSample> shows, dvv::media::Rational fps);
  [[nodiscard]] double percentile(std::vector<double> values, double p);   // nearest rank; 0 if empty
  [[nodiscard]] double median(std::vector<double> values);   // mean of the middle pair; 0 if empty
  }
  ```
- The `perf` JSON line (one line on stdout, exit 0):
  `{"targetFps":…,"actualFps":…,"stalls":…,"bottleneck":"…","frames":600,"presented":…,"overwritten":…,"p99TickLatenessMs":…,"inflight":…,"seekMedianMs":…,"seekMaxMs":…}`.

**Behaviour:**
- `perf` is not a CTest entry (manual, §8.4). It maximises the frame (a realistic compositor extent),
  waits for the stream and two presented frames (30 s), then warm-up: `inflight == 0 && readyAhead ≥ 2`
  for three consecutive ticks (the prefetch window W is full; 60 s), turns Loop on, PlayPause, and
  counts 601 shows (the first starts the clock; 600 intervals; 120 s), pauses, then does 20 seeks to
  random indices (`std::mt19937_64` seed 1, uniform over `[0, length − 1]`, never the current
  position; 10 s each). Seek time = from the `Session::seek` call to the `ShownSet` of the target.
- `stalls` = `status().player.stalls` delta over the play step; `presented` = delta of
  `framesPresented`; `overwritten` = delta of `DisplayState::stats().overwritten`; `bottleneck` = the
  last non-empty `status().bottleneck` seen while playing, else `none`; `p99TickLatenessMs` and
  `actualFps` from `showTiming` over the 601 samples; `inflight` = K.
- **E7 verdict** (applied by hand in Step 7): pass when `actualFps ≥ 59.5`, `stalls == 0` and
  `overwritten ≤ 6` (1 % of 600). `seekMaxMs > 250` → Task C runs. A miss is recorded with the named
  bottleneck and reported to the controller for the arbiter (it does not block M3 silently, E7).

**Code** (`showTiming`, binding; the scenario follows the same structure as Task 10's `player`):

```cpp
ShowTiming showTiming(std::span<const ShowSample> shows, dvv::media::Rational fps) {
    ShowTiming timing;
    if (shows.size() < 2 || fps.num <= 0 || fps.den <= 0) {
        return timing;
    }
    using Ms = std::chrono::duration<double, std::milli>;
    const double period = 1000.0 * static_cast<double>(fps.den) / static_cast<double>(fps.num);
    std::vector<double> lateness;
    lateness.reserve(shows.size() - 1);
    Clock::time_point anchor = shows.front().at;
    std::int64_t steps = 0;  // shows since the anchor; no accumulated rounding
    for (std::size_t i = 1; i < shows.size(); ++i) {
        ++steps;
        const double late = Ms(shows[i].at - anchor).count() - static_cast<double>(steps) * period;
        if (late > period) {  // a stall: PlayerLogic re-anchors at the late show
            ++timing.reanchors;
            anchor = shows[i].at;
            steps = 0;
            continue;
        }
        lateness.push_back(std::max(late, 0.0));
    }
    const double seconds = std::chrono::duration<double>(shows.back().at - shows.front().at).count();
    timing.actualFps = seconds > 0 ? static_cast<double>(shows.size() - 1) / seconds : 0;
    timing.p99LatenessMs = percentile(std::move(lateness), 99.0);
    return timing;
}
```

`perf` scenario (complete):

```cpp
std::vector<SelfTestStep> perfScenario(SelfTest& test) {
    MainFrame& frame = test.frame();
    ShowHistory& history = *test.app().showHistory();
    struct State {
        int quietTicks = 0;
        std::size_t mark = 0;
        std::uint64_t presented0 = 0, overwritten0 = 0, stalls0 = 0;
        std::uint64_t presented = 0, overwritten = 0, stalls = 0;
        std::string bottleneck = "none";
        ShowTiming timing;
        std::mt19937_64 random{1};
        std::int64_t target = 0;
        Clock::time_point seekStart{};
        std::vector<double> seekMs;
    };
    auto st = std::make_shared<State>();
    auto status = [&frame] { return frame.session().status(); };
    std::vector<SelfTestStep> steps;
    using std::chrono::seconds;

    steps.push_back(step("open and maximise", [&frame] { frame.Maximize(); },
        [&frame](std::string& seen) {
            seen = std::format("streams {}, presented {}", frame.streams().size(), presented(frame.viewport()));
            return frame.streams().size() == 1 && presented(frame.viewport()) >= 2;
        }, seconds(30)));
    steps.push_back(step("warm-up", {},
        [st, status](std::string& seen) {
            const SessionStatus s = status();
            seen = std::format("inflight {}, readyAhead {}", s.inflight, s.readyAhead);
            st->quietTicks = s.inflight == 0 && s.readyAhead >= 2 ? st->quietTicks + 1 : 0;
            return st->quietTicks >= 3;  // W is full: nothing in flight, nothing more to fetch
        }, seconds(60)));
    steps.push_back(step("play 600 frames",
        [&frame, &history, st, status] {
            if (!status().player.loop) frame.execute(Command::Loop);
            st->stalls0 = status().player.stalls;
            st->presented0 = presented(frame.viewport());
            st->overwritten0 = frame.viewport().display()->stats().overwritten;
            st->mark = history.size();
            frame.execute(Command::PlayPause);
        },
        [&frame, &history, st, status](std::string& seen) {
            const SessionStatus s = status();
            if (!s.bottleneck.empty()) st->bottleneck = s.bottleneck;
            const std::size_t shown = history.size() - st->mark;
            seen = std::format("{} of 601 shows", shown);
            if (shown < 601) return false;
            frame.execute(Command::PlayPause);  // pause at once: the counters below end here
            st->stalls = status().player.stalls - st->stalls0;
            st->presented = presented(frame.viewport()) - st->presented0;
            st->overwritten = frame.viewport().display()->stats().overwritten - st->overwritten0;
            std::vector<ShowSample> samples;
            for (const ShownSet& set : history.since(st->mark)) {
                if (samples.size() == 601) break;
                samples.push_back({set.index, set.at});
            }
            st->timing = showTiming(samples, frame.streams().at(0).info.fps);
            return true;
        }, seconds(120)));
    for (int k = 1; k <= 20; ++k) {
        steps.push_back(step(std::format("seek {}", k),
            [&frame, &history, st, status] {
                const SessionStatus s = status();
                std::uniform_int_distribution<std::int64_t> pick(0, s.player.length.value_or(1) - 1);
                do {
                    st->target = pick(st->random);
                } while (st->target == s.player.position && s.player.length.value_or(1) > 1);
                st->mark = history.size();
                st->seekStart = Clock::now();
                frame.session().seek(st->target);
            },
            [&history, st](std::string& seen) {
                for (const ShownSet& set : history.since(st->mark)) {
                    if (set.index == st->target) {
                        st->seekMs.push_back(
                            std::chrono::duration<double, std::milli>(set.at - st->seekStart).count());
                        return true;
                    }
                }
                seen = std::format("target {} not shown", st->target);
                return false;
            }, seconds(10)));
    }
    steps.push_back(step("report",
```

### Task C (conditional): `stop_token` cancellation

**Run only if** Task 11b recorded `seekMaxMs > 250` on the 4K file (X4, R7). Otherwise skip it: X4

**Goal:** Stale prefetch requests stop occupying the in-flight cap and stop their conversion at the
next step, so a seek is not queued behind up to K stale 4K conversions: the M2 D10 `stop_token` hook
in `FrameStore::acquire`, used by the Session.

**Files:**
- dvv-media: `include/dvv/media/frame_store.h` (the `acquire` overload), `src/frame_store.cpp` and
  the store's internal pending-entry/waiter structures (M2 Task 7a's files), new
  `tests/test_frame_store_cancel.cpp`, `tests/CMakeLists.txt`, `README.md` (one paragraph).
- rawreel: `src/core/player.h/.cpp` (`requestCancelled`), `src/core/session.cpp`, the player unit-test
  file (`tests/test_player.cpp`) and the Session integration tests (Task 5's file); Task 5's `slow:`
  source gains `count=C` (default 1): indices `index .. index+C−1` wait on the gate
  (`tests/test_source_provider.cpp`).

**Interfaces:**
- Consumes: M2's `FrameStore` (single-flight pending entries with removable waiter nodes and a count
  of waiters on pending creations — the D10 hook), `Errc::Cancelled` (dvv-base); M3 Tasks 4, 5 and 6
  (`PlayerLogic`, the Session's request tasks, tags `{streamSet, epoch, generation}` where
  `generation` is the out-of-memory window generation, `spawn`).
- Produces:
  ```cpp
  // frame_store.h (addition; the old overload forwards with an empty token)
  // As acquire() above; additionally, once `stop` is requested the call completes with
  // Errc::Cancelled "the request was cancelled" (posted to resume_on, never inline) unless the
  // lease was already handed over. A cancelled request never returns a lease.
  [[nodiscard]] Task<Result<ReadLease>> acquire(
      const FrameKey& key, const ViewSpec& want, IExecutor& resume_on, std::stop_token stop,
      std::source_location where = std::source_location::current());

  // player.h (addition): drops the outstanding mark of `index` without readiness (the cap frees and
  // the index may be requested again later); no effect when it is not outstanding.
  void requestCancelled(std::int64_t index);
  ```

**Behaviour (binding):**
- **Store.**
  1. Stop requested before the call → `Cancelled`, posted to `resume_on`; no planning, no produce.
  2. A waiter on a pending entry registers a `std::stop_callback`. The callback takes the store
     mutex; if the waiter is still registered, it unlinks the node (the D10 removable node),
     decrements the pending creation's interest count, stores `Cancelled`, releases the mutex and
     posts the continuation to the waiter's `resume_on`. Completion takes the same mutex and marks
     each node resumed; whichever runs first wins; each waiter is resumed exactly once.
  3. A pending creation's interest = its creator's request + its registered waiters, minus the
     cancelled ones. The creator checks the interest under the mutex **between steps** only (a running
     produce, CPU conversion or GPU submission is never interrupted; its output is published). When
     the interest is 0, the remaining pending entries of the plan fail with `Cancelled` (not cached in
     the no-path cache, not counted in `failures`) and the creator returns `Cancelled`.
  4. A creator whose own token is stopped while other waiters remain continues for them and returns
     `Cancelled` to its own caller at the end.
  5. The `stop_callback` lives in the coroutine frame from registration until the waiter is resumed
     (destroyed before the frame continues), so no callback outlives its node.
- **Session.** Each request task gets a `std::stop_source`; every stream's `acquire` in its `whenAll`
  takes the same token. The Session keeps `index → stop_source` for outstanding requests. After every
  `pump()`, an outstanding index with `!player.keeps(index)` or a stale tag (`currentTag()`, i.e.
  `{streamSet, epoch, generation}`) gets
  `request_stop()`, `player.requestCancelled(index)` and is forgotten. A `Cancelled` result is handled
  as a stale result (dropped, leases released, no `frameFailed`, no `error`). The `spawn` in-flight
  counter still counts the task until it ends, so `shutdown()` waits for it as before.

**Code (the waiter cancel handshake; adapt to M2 Task 7a's node and field names):**

```cpp
// Waiter path inside acquire(..., stop), after "register waiter on pending entry p" (store mutex held):
//   node.resumeOn = &resume_on; node.handle = <this coroutine>; node.result = &result;
//   p.waiters.push_back(node); ++p.interest;
// then, still before suspending:
std::optional<std::stop_callback<CancelWaiter>> on_stop;
if (stop.stop_possible()) {
    on_stop.emplace(stop, CancelWaiter{this, &p, &node});  // may run at once on this thread
}
co_await suspend;  // resumed by completion or by CancelWaiter, exactly once
on_stop.reset();   // waits for a running callback (it only takes the mutex briefly and posts)

struct CancelWaiter {
    FrameStore::Impl* store; PendingEntry* pending; WaiterNode* node;
    void operator()() const noexcept {
        std::unique_lock lock(store->mutex);
        if (node->resumed) {
            return;                        // completion already handed it a result
        }
        node->resumed = true;
        pending->waiters.erase(*node);     // removable node (M2 D10 hook)
        --pending->interest;               // the creator checks it between steps
        *node->result = dvv::makeError(dvv::Errc::Cancelled, "the request was cancelled");
        lock.unlock();
        node->resumeOn->post(node->handle);  // never inline (M2 D12)
    }
};
// Completion (store mutex held): for each node in p.waiters with !node.resumed: resumed = true,
// fill its result, collect it; post the collected continuations after unlocking.
```

(If constructing the `stop_callback` while the store mutex is held could run `CancelWaiter` inline and
self-deadlock, register it after unlocking and re-check `node->resumed` under the lock: the callback
handles an already-completed node by returning.)
